match /staff/{uid} {
  allow read: if request.auth != null;

  // self-register your own pending record only, with no real role
  allow create: if isStaffAdmin() ||
    (request.auth != null && request.auth.uid == uid
     && request.resource.data.role == 'none');

  // self-edit is allowed EXCEPT the founder-controlled fields; admin can do all
  allow update: if isStaffAdmin()
                || (request.auth != null && request.auth.uid == uid
                    && !request.resource.data.diff(resource.data).affectedKeys()
                          .hasAny(['role','docs','pay','staffId','title','resourceRole','status']));

  allow delete: if isStaffAdmin();

  match /logs/{day} {
    allow read: if request.auth != null && (request.auth.uid == uid || isStaffAdmin());
    allow create, update: if request.auth != null && request.auth.uid == uid;
    allow delete: if isStaffAdmin();
  }
  match /sessions/{day} {
    allow read: if request.auth != null && (request.auth.uid == uid || isStaffAdmin());
    allow create, update: if request.auth != null && request.auth.uid == uid;
    allow delete: if isStaffAdmin();
  }
  match /claims/{cid} {
    allow read: if request.auth != null && (request.auth.uid == uid || isStaffAdmin());
    allow create: if request.auth != null && request.auth.uid == uid
                  && request.resource.data.status == 'pending';
    allow update, delete: if isStaffAdmin();
  }
}
